        function renderQuiz() {
            const container = document.getElementById('quiz-container');
            container.innerHTML = '';

            const filtered = currentFilter === 'all' 
                ? questionsData 
                : questionsData.filter(q => q.level === currentFilter);

            filtered.forEach((q, index) => {
                const isAnswered = userAnswers.hasOwnProperty(q.id);
                const selectedIdx = userAnswers[q.id];

                let levelBadge = '';
                if(q.level === 'A') levelBadge = '<span class="bg-emerald-100 text-emerald-800 text-xs px-2.5 py-1 rounded-md font-bold">A деңгей (Жеңіл)</span>';

        // FEUDAL HIERARCHY DRAG & DROP LOGIC
        function setupPyramidDragAndDrop() {
            const items = document.querySelectorAll('.drag-item');
            const slots = document.querySelectorAll('.pyramid-slot');

            items.forEach(item => {
                item.addEventListener('dragstart', (e) => {
                    e.dataTransfer.setData('text/plain', e.target.id);
                });
            });

            slots.forEach(slot => {
                slot.addEventListener('dragover', (e) => {
                    e.preventDefault();
                    slot.classList.add('drag-over');
                });

                slot.addEventListener('dragleave', () => {
                    slot.classList.remove('drag-over');
                });

                slot.addEventListener('drop', (e) => {
                    e.preventDefault();
                    slot.classList.remove('drag-over');
                    const itemId = e.dataTransfer.getData('text/plain');
                    const draggedEl = document.getElementById(itemId);
                    if (draggedEl) {
                        slot.innerHTML = '';
                        slot.appendChild(draggedEl);
                    }
                });
            });
        }

        function checkPyramid() {
            let correct = 0;
            const slots = document.querySelectorAll('.pyramid-slot');
            slots.forEach(slot => {
                const targetRank = slot.getAttribute('data-rank');
                const child = slot.querySelector('.drag-item');
                if (child) {
                    const itemRank = child.getAttribute('data-rank');
                    if (targetRank === itemRank) {
                        correct++;
                        slot.classList.add('bg-emerald-500/30', 'border-emerald-400');
                    } else {
                        slot.classList.add('bg-rose-500/30', 'border-rose-400');
                    }
                }
            });

            const feedback = document.getElementById('pyramid-feedback');
            feedback.classList.remove('hidden');
            if (correct === 5) {
                feedback.className = "mt-4 p-4 rounded-xl font-bold text-sm text-center bg-emerald-100 text-emerald-800 border border-emerald-300";
                feedback.innerHTML = "🎉 Тамаша! Феодалдық сатының (иерархияның) барлық 5 деңгейін дұрыс құрастырдыңыз!";
                confetti({ particleCount: 80, spread: 60 });
            } else {
                feedback.className = "mt-4 p-4 rounded-xl font-bold text-sm text-center bg-amber-100 text-amber-800 border border-amber-300";
                feedback.innerHTML = `⚠️ Дұрысы: ${correct} / 5 саты. Түстер арқылы қай сатыда қателік бар екенін қарап, түзетіңіз.`;
            }
        }

        // MATCHING GAME LOGIC
        function renderMatching() {
            const board = document.getElementById('matching-board');
            board.innerHTML = '';

            // Shuffle terms and definitions
            const terms = [...matchingData].sort(() => Math.random() - 0.5);
            const defs = [...matchingData].sort(() => Math.random() - 0.5);

            let termsCol = '<div class="space-y-3"><h4 class="font-bold text-slate-800 mb-2">Терминдер:</h4>';
            terms.forEach(t => {
                termsCol += `
                    <div class="p-4 bg-white rounded-xl border border-slate-200 font-bold text-indigo-900 shadow-sm flex items-center justify-between">
                        <span>${t.term}</span>
                        <select id="select-${t.id}" class="text-xs p-2 rounded-lg border border-slate-300 bg-slate-50 font-normal">
                            <option value="">-- Анықтаманы таңдаңыз --</option>
                            ${defs.map((d, i) => `<option value="${d.id}">${i + 1}. ${d.def.substring(0, 35)}...</option>`).join('')}
                        </select>
                    </div>
                `;
            });
            termsCol += '</div>';

            let defsCol = '<div class="space-y-3"><h4 class="font-bold text-slate-800 mb-2">Анықтамалар тізімі:</h4>';
            defs.forEach((d, idx) => {
                defsCol += `
                    <div class="p-4 bg-indigo-50 rounded-xl border border-indigo-100 text-xs text-indigo-950">
                        <strong class="text-indigo-700 text-sm mr-2">${idx + 1}.</strong> ${d.def}
                    </div>
                `;
            });
            defsCol += '</div>';

            board.innerHTML = termsCol + defsCol;
        }

        function checkMatching() {
            let correct = 0;
            matchingData.forEach(item => {
                const sel = document.getElementById(`select-${item.id}`);
                if (sel && sel.value === item.id) {
                    correct++;
                    sel.classList.add('border-emerald-500', 'bg-emerald-50');
                } else if (sel) {
                    sel.classList.add('border-rose-500', 'bg-rose-50');
                }
            });

            const feedback = document.getElementById('matching-feedback');
            feedback.classList.remove('hidden');
            if (correct === matchingData.length) {
                feedback.className = "mt-4 p-4 rounded-xl font-bold text-sm text-center bg-emerald-100 text-emerald-800 border border-emerald-300 max-w-md mx-auto";
                feedback.innerHTML = "✨ Өте тамаша! Барлық терминдер мен анықтамалар дұрыс сәйкестендірілді!";
                confetti({ particleCount: 70, spread: 60 });
            } else {
                feedback.className = "mt-4 p-4 rounded-xl font-bold text-sm text-center bg-amber-100 text-amber-800 border border-amber-300 max-w-md mx-auto";
                feedback.innerHTML = `Нәтиже: ${correct} / ${matchingData.length} дұрыс. Түзетіп қайта тексеріңіз.`;
            }
        }

### Веб-қосымшаның мүмкіндіктері мен құрылымы:

1. **Сараланған тапсырмалар блогы (A, B, C деңгейлері):**
   - **А деңгейі (Көрнекі-танымдық):** Негізгі терминдер, тұлғалар (Флавио Бьондо) және негізгі фактілер.
   - **В деңгейі (Түсіну мен қолдану):** «Барщина» мен «Оброк», сервтер мен вилландар сияқты тарихи ұғымдарды салыстыру.
   - **С деңгейі (Талдау мен жинақтау):** Себеп-салдарлық байланыстар, Шығыс пен Еуропадағы билік жүйесін аналитикалық талдау.

2. **Интерактивті Феодалдық саты симуляторы:**
   - Оқушылар «Король -> Герцогтер/Графтар -> Барондар -> Рыцарьлар -> Шаруалар» иерархиясын сүйреп апару (Drag & Drop) арқылы дұрыс ретпен орналастырып, тез арада тексере алады.

3. **Терминдер мен Ұғымдар ойыны:**
   - Бенефиция, Оммаж, Донжон, Вотчина сияқты ортағасырлық маңызды терминдерді өз анықтамасымен жұптастыру.
